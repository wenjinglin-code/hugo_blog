MMU

MMU 硬件原理（priv-isa ）

页表目录：satp , 结构如下（大端）

PPN: 一级页表的 physical page number

ASID：页表的标识号（软件配置），用于和 TLB 中的条目匹配，这样就可以达到 切换 进程时，不刷新 tlb 的效果（每个进程对应一个 ASID)

asid 的位数和 hw 有关，不同硬件 asid bit 不一样，可以通过写 0xFFFF 到 ASID，然后回读，确认 bit 数

MODE：标识是几级页表，虚拟地址长度

  • SV32（没有一些特殊属性（大页，页表类型））（32bits va 转换为 34bits pa）

  • SV57

如果实现了 svadu 扩展，硬件会自动更新 pte.D/A bit，就不需要在 page-fault 中处理 D/A bit

没有实现 svadu，需要实现 svade 扩展，在以下情况触发异常，然后再物理页中更新 D/A bit：

  • 访问物理页并且 A==0；
  • 写物理页并且 D==0；

D bit 会在回写完毕后，置 0；A bit 会在页表回收机制中，置 0；

U bit 用于控制该页面是否可以被 u-mode 访问，0:表示不可以访问，1:表示可以被访问，U=1 的时候，s-mode 也不能访问该 page，除非 sstatus.SUM = 1

大页支持

page_size == PMD_SIZE / PUD_SIZE

N 位好像只是告诉 MMU 这个 pte 是连续的pte，如果 page_size == PMD_SIZE 直接配置 pmd_t 为 pte 就可以了，对应的 pte.ppn[0] 要配置为 0

page_size != PMD_SIZE / PUD_SIZE

需要配置 N 位，告诉 MMU 这个 pte 是**一个 NAPOT 范围**的一部分

目前只规定了 64k，page offset 怎么配套，没拿到 pte 前又不知道 pte 是大页？

spec 有说明翻译的流程：

大体架构：

riscv / arm64 大页的意思，都是使用连续的 ptes 映射大 size 的 page，每个 pte 映射的还是 4k，只是 tlb 可以将多个 pte 映射，只用一个 entry 表示 va 到 大 size page 的映射，减少大内存映射对 tlb entry 的使用

powerpc 的大页实现，是直接配置 psize 到 hpte

riscv 大页新的patch：[PATCH RESEND v2 0/9] Merge arm64/riscv hugetlbfs contpte support - Alexandre Ghiti (kernel.org)

34// arm64 的大页实现，也是配置 entry
35pte_t arch_make_huge_pte(pte_t entry, unsigned int shift, vm_flags_t flags)
36{
37	size_t pagesize = 1UL << shift;
38
39	entry = pte_mkhuge(entry);
40	if (pagesize == CONT_PTE_SIZE) {
41		entry = pte_mkcont(entry);
42	} else if (pagesize == CONT_PMD_SIZE) {
43		entry = pmd_pte(pmd_mkcont(pte_pmd(entry)));
44	} else if (pagesize != PUD_SIZE && pagesize != PMD_SIZE) {
45		pr_warn("%s: unrecognized huge page size 0x%lx\n",
46			__func__, pagesize);
47	}
48	return entry;
49}

44/*
45 * Task size is 0x4000000000 for RV64 or 0x9fc00000 for RV32.
46 * Note that PGDIR_SIZE must evenly divide TASK_SIZE.
47 * Task size is:
48 * -        0x9fc00000	(~2.5GB) for RV32.
49 * -      0x4000000000	( 256GB) for RV64 using SV39 mmu
50 * -    0x800000000000	( 128TB) for RV64 using SV48 mmu

55 * "load and store effective addresses, which are 64bits, must have bits

内核页表初始化流程

early_pg_dir、trampoline_pg_dir、 swapper_pg_dir 的作用

代码流程

 1_start_kernel
 2|- setup_vm // 建立早期的页表目录
 3	 |- set_satp_mode(dtb_pa); // 检查支持的 mmu 模式，并配置 satp_mode，对应 STAP:MODE
 4	 |- create_pgd_mapping(early_pg_dir, FIXADDR_START,
 5		   fixmap_pgd_next, PGDIR_SIZE, PAGE_TABLE); // **early_pg_dir** 中为 fixmap 创建页表，没创建 pte，用于 early_ioremap
 6	 |- create_pgd_mapping(trampoline_pg_dir, kernel_map.virt_addr,
 7		   trampoline_pgd_next, PGDIR_SIZE, PAGE_TABLE); // **trampoline_pg_dir** 创建内核的页表映射, 只映射了前面的 2M，看代码，是用来第一次使能 mmu，va != pa 导致异常，是使pc 变为 va，用到的中间页表
 8	 |- create_kernel_page_table(early_pg_dir, true); // **early_pg_dir** 目录的 kernel 映射
 9	 |- create_fdt_early_page_table(__fix_to_virt(FIX_FDT), dtb_pa); // **early_pg_dir** 目录中，fixmap 虚拟空间中，建立 dtb 映射（**dtb_early_va -> dtb_pa**）
10|- relocate_enable_mmu // 第一次用 **trampoline_pg_dir** ，触发异常后进入 CSR_TVEC 中配置的虚拟地址，然后再次切换到 **early_pg_dir**
11
12setup_arch
13|- paging_init
14	 |- setup_bootmem(); // 保留内存到 memblock
15		  |- kernel_map.**va_pa_offset** = PAGE_OFFSET - phys_ram_base; // physical addr = virtual addr - va_pa_offset = virtual addr - PAGE_OFFSET + phys_ram_base 
16	 |- setup_vm_final
17		  |- create_pgd_mapping(swapper_pg_dir, FIXADDR_START, __pa_symbol(fixmap_pgd_next), PGDIR_SIZE, PAGE_TABLE); // **swapper_pg_dir** 中为 fixmap 创建页表，没创建 pte，用于 early_ioremap
18		  |- create_linear_mapping_page_table(); // **swapper_pg_dir** 中对所有没使用的内存和 kernel 进行到 PAGE_OFFSET 的线性映射
19		  |- create_kernel_page_table(swapper_pg_dir, false); // **swapper_pg_dir** 中对kernel 到 kernel_mapping.virt_addr 的映射
20			|- csr_write(CSR_SATP, PFN_DOWN(__pa_symbol(swapper_pg_dir)) | satp_mode); // 切换页表目录，使用 swapper_pg_dir 作为 init_mm.pgd（初始值就是 swapper_pg_dir）

asid 分配

isa

最新的 isa 说明 asid 是 per cpu 私有的，后面可能会扩展 全局的asid

但是旧的版本 建议软件共用 asid，而且现在最新的代码也是共用 asid 的，后面可以提一下 patch

代码流程

 1// 相关变量  
 2&per_cpu(active_context, cpu)  // 表示当前正在用的 asid
 3&per_cpu(reserved_context, cpu) // 更新 asid 版本时，保留的上一个正在用的asid
 4
 5// 早期启动阶段，配置的
 6num_asids = 1 << asid_bits;
 7asid_mask = num_asids - 1;
 8atomic_long_set(&current_version, num_asids); // asid 的初始版本
 9context_asid_map = bitmap_zalloc(num_asids, GFP_KERNEL); // asid 的分配map
10
11// 进程切换
12__schedule
13|- rq = context_switch(rq, prev, next, &rf);
14   |- switch_mm_irqs_off(prev->active_mm, next->mm, next); // switch_mm
15		  |- set_mm(prev, next, cpu);
16				 |- set_mm_asid(next, cpu); // 配置 asid，asid 都是在轮到进程运行的时候，才开始初始化
17						|- cntx = __new_context(mm); // **mm->context.id (1 << asid_bits | asid)** 没有被配置 asid，分配新的 asid
18						|	 |- if (cntx != 0) // 已经分配过 asid 但是版本不匹配，才到这
19						|	 |  |- unsigned long newcntx = ver | (cntx & asid_mask);
20						|	 |	|- if (check_update_reserved_context(cntx, newcntx)) return newcntx; 
21						|	 |	|	 |- if (per_cpu(**reserved_context**, cpu) == cntx) { hit = true; per_cpu(reserved_context, cpu) = newcntx;} // 如果 cntx 是更新 asid 版本时，**保留的上一个正在用的asid，则不用分配，已经占位了** 
22						|	 |	|- if (!__test_and_set_bit(cntx & asid_mask, context_asid_map)) return newcntx; // 如果新版本asid，这个 asid 没被用，可以直接用
23						|	 |	|- asid = find_next_zero_bit(context_asid_map, num_asids, cur_idx); // 查找没有用的asid
24						|	 | 	|- ver = atomic_long_add_return_relaxed(num_asids, &current_version); // 如果 asid 用完了，更新版本，current_version = current_version + num_asids;
25						|	 |	|- __flush_context(); // 去除原有的asid
26						|	 |	|	 |- bitmap_zero(context_asid_map, num_asids); // 清空 context_asid_map
27						|	 |	|	 |- for_each_possible_cpu(i) 
28						|	 |	|	 |  |- cntx = atomic_long_xchg_relaxed(&per_cpu(active_context, i), 0); // old_active_context = active_context; active_context = 0; return old_active_context;
29						|	 |	|	 |  |- __set_bit(cntx & asid_mask, context_asid_map);
30						|	 |	|	 |  |- per_cpu(**reserved_context**, i) = cntx; // **保留上个版本的 active_context** 
31						|	 |	|	 |- __set_bit(0, context_asid_map); // 保留 init_mm (kernel) 用的 asid
32						|	 |	|	 |- cpumask_setall(&**context_tlb_flush_pending**); // 配置 context_tlb_flush_pending ，**需要刷新 tlb** 
33						|	 |  |- asid = find_next_zero_bit(context_asid_map, num_asids, 1);
34						|	 |  |- return asid | ver; // 返回 cntx
35						|- if (cpumask_test_and_clear_cpu(cpu, &context_tlb_flush_pending)) need_flush_tlb = true; // **是否需要刷新 tlb** 
36						|- atomic_long_set(&mm->context.id, cntx); // **mm->context.id** 填充 
37						|- csr_write(CSR_SATP, virt_to_pfn(mm->pgd) | ((cntx & asid_mask) << SATP_ASID_SHIFT) | satp_mode); // 更新 stap
38						|- if (need_flush_tlb) local_flush_tlb_all(); // **根据需要刷新 tlb，更新过 asid 版本才需要刷新，所以可以多个进程的 tlb 同时存在** 
39							 |- __asm__ __volatile__ ("sfence.vma" : : : "memory"); // 刷掉所有的 tlb
