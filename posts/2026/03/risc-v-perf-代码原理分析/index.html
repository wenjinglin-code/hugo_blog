2. perf 用户态工具

3.4 采样周期设置原理 (riscv_pmu_event_set_period)

这是 Perf 的核心算法，用于实现采样中断功能。在 riscv_pmu_start() 中调用。

3.4.1 核心数据结构

 1struct hw_perf_event {
 2    u64 sample_period;      // 期望的采样周期（期望的目标计数，计数达到此值时触发中断）
 3    u64 last_period;        // 上一次的采样周期（上一次的目标计数）
 4    u64 period_left;        // 距离目标计数还剩多少计数，即 `sample_period - prev_count`
 5                            // 其负数作为 counter 的初始值，这样增加计数，需要计数到 2period_left 才到达 sample_period
 6    u64 prev_count;         // 上一次读取的计数器值（软件记录）
 7    // ...
 8};
 9
10struct perf_event {
11    // ...
12	local64_t			count; // 整个采样过程已经记录的 count 数
13    // ...
14}

3.4.2 算法详解

 1int riscv_pmu_event_set_period(struct perf_event *event)
 2{
 3    struct hw_perf_event *hwc = &event->hw;
 4    s64 left = local64_read(&hwc->period_left);  // 读取剩余计数
 5    s64 period = hwc->sample_period;              // 目标采样周期(目标计数)
 6    int overflow = 0;
 7    uint64_t max_period = riscv_pmu_ctr_get_width_mask(event);  // 计数器位掩码，最大计数值
 8
 9    // left 有效范围为 [0, 2period]，超过范围的情况要重新配置 period_left
10    // 情况1: left 已经是负数且绝对值大于等于 period
11    // 说明之前已经溢出了 period 个数，需要重置 period_left
12    if (unlikely(left <= -period)) {
13        left = period;
14        local64_set(&hwc->period_left, left);
15        hwc->last_period = period;
16        overflow = 1;
17    }
18
19    // 情况2: left <= 0，说明之前发生了溢出但不够严重
20    if (unlikely(left <= 0)) {
21        left += period;
22        local64_set(&hwc->period_left, left);
23        hwc->last_period = period;
24        overflow = 1;
25    }
26
27    // 限制最大周期，防止计数器值超过要设置的值
28    // 考虑中断延迟，保守地设置为宽度的一半
29    if (left > (max_period >> 1))
30        left = (max_period >> 1);
31
32    // ★ 核心技巧: 使用负数作为初始值
33    // prev_count 会写入硬件计数器 CSR
34    local64_set(&hwc->prev_count, (u64)-left);
35
36    perf_event_update_userpage(event);
37    return overflow;
38}

3.4.3 负数初始值技巧

这是 Perf 实现采样的关键技巧：

  1. 假设 left = 1000（剩余要计数 1000 次触发中断）
  2. 设置 prev_count = -1000（补码表示为 0xFFFFFC18）
  3. 硬件行为：
    • 计数器从 0xFFFFFC18 开始递增：0xFFFFFC18 → 0xFFFFFC19 → … → 0
    • 继续递增：0 → … → 1000
  4. 实际计数：从 -1000 到 0 是 1000，从 0 到 1000 也是 1000，总体计数就可以到 2000 了
  5. 触发中断：当计数达到 1000 时溢出，触发 perf 采样中断

3.4.4 计数器读取 (riscv_pmu_event_update)

在计数器停止或读取时调用，计算增量：

 1u64 riscv_pmu_event_update(struct perf_event *event)
 2{
 3    // 使用 CAS 确保原子性，防止并发读取
 4    do {
 5        prev_raw_count = local64_read(&hwc->prev_count);
 6        new_raw_count = rvpmu->ctr_read(event);  // 读取硬件 CSR
 7    } while (local64_cmpxchg(&hwc->prev_count, prev_raw_count, new_raw_count) != prev_raw_count);
 8
 9    // 计算增量（处理溢出回绕）
10    delta = (new_raw_count - prev_raw_count) & cmask;
11
12    // 更新统计
13    local64_add(delta, &event->count);        // 总计数
14    local64_sub(delta, &hwc->period_left);   // 剩余计数
15
16    return delta;
17}

6. 关键数据结构

6.1 perf_event_attr (用户态定义事件)

 1struct perf_event_attr {
 2    __u32 type;           // 事件类型 (HARDWARE/SOFTWARE/RAW...)
 3    __u64 config;         // 事件配置，
 4                          // type 为 hw，该值为 PERF_COUNT_HW_CPU_CYCLES...
 5                          // type 为 raw，bits[47:0] 为 event_data
 6    __u64 config1;        // 扩展配置1
 7    __u64 config2;        // 扩展配置2
 8    __u64 sample_period;  // 采样周期
 9    // ... 更多字段
10};

6.2 PMU 事件映射

1// OpenSBI 中事件索引编码
2// event_idx = (type << SBI_PMU_EVENT_IDX_TYPE_OFFSET) | code
3// 其中 SBI_PMU_EVENT_IDX_TYPE_OFFSET = 12
4//
5// type: SBI_PMU_EVENT_TYPE_HW (0x0)           - 硬件事件 (如 cycles, instructions)
6//       SBI_PMU_EVENT_TYPE_HW_CACHE (0x1)      - 缓存事件 (如 L1D, L1I)
7//       SBI_PMU_EVENT_TYPE_HW_RAW (0x2)        - 原始硬件事件 (v1)
8//       SBI_PMU_EVENT_TYPE_HW_RAW_V2 (0x3)     - 原始硬件事件 (v2, 64位 event_data)
9//       SBI_PMU_EVENT_TYPE_FW (0xf)            - 固件事件 (如 firmware counter)

6.3 计数器信息

 1// OpenSBI 返回的计数器信息
 2union sbi_pmu_ctr_info {
 3    unsigned long value;
 4    struct {
 5        unsigned long csr:12;    // CSR 编号
 6        unsigned long width:6;   // 计数器宽度
 7        unsigned long reserved;
 8        unsigned long type:1;    // 0=HW, 1=FW
 9    };
10};

7. 适配自定义事件（raw type event）

目前的代码中没有适配 PERF_TYPE_RAW 事件的添加，即不支持 SBI_PMU_EVENT_TYPE_RAW，需要用户自行添加 sysfs 文件节点 或者 在 tools/perf/pmu-events/arch/riscv/ 中添加 pmu 的 json 文件。

7.1 通过添加 sysfs 事件节点

 1// 1. 添加 event 文件节点 wjl_test，输出内容为 "event=0x2000d"
 2PMU_EVENT_ATTR_STRING(wjl_test, event_attr_test, "event=0x2000d");
 3	/* 
 4		等效于 firmware=0x0,event=0x2000d
 5		event 表示填充到 attr.config.event 的值, 这个值要和 dts 中的 riscv,raw-event-to-mhpmcounters 对应
 6		可以通过 -e wjl_test 选用事件
 7		也可以通过 -e cpu/event=0x2000d,firmware=0x0/[cpu/wjl_test/] 选用事件
 8		
 9		因为 /sys/bus/event_source/devices/cpu/format 下：
10		event = "config:0-47", firmware = "config:62-63"，因此：
11		- `event=0x2000d` 被解析到 `perf_event_attr.config` 的 bits[0-47]
12		- `firmware=0x0` 被解析到 `perf_event_attr.config` 的 bits[62-63]
13	 */
14

17	NULL,
18};

22	.name = "events",
23	.attrs = riscv_arch_events_attr,
24};

7.3 支持 SBI_PMU_EVENT_TYPE_HW_RAW_V2

drivers/perf/riscv_pmu_sbi.c 并没有支持 SBI_PMU_EVENT_TYPE_HW_RAW_V2，需要做以下修改：

  1. 修改 format 定义，扩展 event 字段占用 attr.config 中的位数：

    1// 原来: PMU_FORMAT_ATTR(event, "config:0-47");
2// 修改为:
3PMU_FORMAT_ATTR(event, "config:0-55");

  2. 修改事件映射接口，适配 56 位 event_data 的位宽：

    1// pmu_sbi_event_map 接口中
2case PERF_TYPE_RAW:
3    *econfig = config & RISCV_PMU_RAW_EVENT_MASK; // RISCV_PMU_RAW_EVENT_MASK 改为支持 56 bits 的mask
4    break;

9. 总结

RISC-V Perf 架构是一个分层设计：

  1. perf 用户态：解析事件名称，验证事件可用性
  2. Linux 内核驱动：管理事件到计数器的映射，处理计数器的分配和读写
  3. OpenSBI 固件：实现 SBI PMU 扩展，管理硬件计数器 CSR，处理事件配置
