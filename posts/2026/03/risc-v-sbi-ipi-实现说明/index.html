RISC-V SBI IPI 实现说明

1. 概述

IPI（Inter-Processor Interrupt） 是处理器间中断，用于在不同 Hart（Hardware Thread，硬件线程） 之间进行通信和同步。在 RISC-V 架构中，没有实现 AIA-IMSIC 时，Linux 内核需要使用 SBI（Supervisor Binary Interface，监管者二进制接口） 提供的接口来触发其他 Hart 的软件中断（各平台自定义实现）。

3. Linux Kernel 侧实现

3.1 IPI 初始化流程

Linux Kernel 通过 Device Tree（设备树）获取中断控制器信息，并初始化 IPI 机制。以下是主要初始化步骤：

 1// arch/riscv/kernel/sbi-ipi.c
 2
 3void __init sbi_ipi_init(void)
 4{
 5	// ...
 6
 7	/*
 8	 * 步骤 1: 向 intc domain 获取 RV_IRQ_SOFT (Supervisor Software Interrupt)
 9	 *         的 Linux 中断号
10	 *         RV_IRQ_SOFT 是 S 模式软件中断的中断号
11	 */
12	sbi_ipi_virq = irq_create_mapping(domain, RV_IRQ_SOFT);
13	if (!sbi_ipi_virq) {
14		pr_err("unable to create INTC IRQ mapping\\n");
15		return;
16	}
17
18	/*
19	 * 步骤 2: 创建 Linux 模拟的 IPI 中断控制器 (ipi_mux_domain)
20	 *         - virq: 该控制器的硬件中断号对应的 Linux 中断号
21	 *         - sbi_send_ipi: 触发中断控制器中断的接口
22	 *         - BITS_PER_BYTE: 支持的 IPI 事件类型数量（8 种）
23	 */
24	virq = ipi_mux_create(BITS_PER_BYTE, sbi_send_ipi);
25
26	// ...
27
28	/*
29	 * 步骤 3: 为 RV_IRQ_SOFT 设置链式中断处理函数
30	 *         当目标 Hart 收到 IPI 时，会调用 sbi_ipi_handle
31	 */
32	irq_set_chained_handler(sbi_ipi_virq, sbi_ipi_handle);
33
34	// ...
35
36	/*
37	 * 步骤 4: 为 virq 注册对应中断处理函数，处理各个 IPI 事件类型
38	 */
39	riscv_ipi_set_virq_range(virq, BITS_PER_BYTE);
40	pr_info("providing IPIs using SBI IPI extension\\n");
41
42	// ...
43}

3.2 IPI 发送流程

 1// arch/riscv/kernel/sbi-ipi.c
 2
 3/*
 4 * Linux 发送 IPI 的入口函数
 5 * 通过 SBI 调用通知目标 Hart
 6 */
 7static void __sbi_send_ipi_v02(unsigned int cpu)
 8{
 9	/*
10	 * 调用 SBI 接口发送 IPI
11	 * SBI_EXT_IPI: IPI 扩展 ID (0x73504900)
12	 * SBI_EXT_IPI_SEND_IPI: 发送 IPI 函数 ID (0x0)
13	 * 1UL: hmask - 目标 Hart 位掩码（只发送到 hbase 对应的 hart）
14	 * cpuid_to_hartid_map(cpu): hbase - 目标 Hart 的起始 Hart ID
15	 *
16	 * ECALL 会触发从 S 模式到 M 模式的切换
17	 */
18	ret = sbi_ecall(SBI_EXT_IPI, SBI_EXT_IPI_SEND_IPI,
19			1UL, cpuid_to_hartid_map(cpu), 0, 0, 0, 0);
20}

4.3 IPI 发送流程

 1// lib/sbi/sbi_ipi.c
 2
 3/*
 4 * 发送 IPI 到多个目标 Hart
 5 *
 6 * @param hmask: 目标 Hart 位掩码
 7 * @param hbase: 目标 Hart 的起始 Hart ID
 8 * @param event: IPI 事件类型（bit 编号）
 9 * @param data: 传递给 IPI 事件处理函数的数据
10 *
11 * @return 0 表示成功，负数表示错误码
12 */
13int sbi_ipi_send_many(ulong hmask, ulong hbase, u32 event, void *data)
14{
15	// ...
16
17	/* 发送 IPI 到每个目标 Hart */
18	do {
19		retry_needed = false;
20		sbi_hartmask_for_each_hartindex(i, &target_mask) {
21			rc = sbi_ipi_send(scratch, i, event, data);
22			// ...
23		}
24	} while (retry_needed);
25
26	/* 同步所有 IPI，确保所有目标 Hart 都收到 IPI */
27	sbi_ipi_sync(scratch, event);
28
29	return rc;
30}
31
32/*
33 * 发送 IPI 到单个远程 Hart
34 *
35 * @param scratch: 当前 Hart 的 scratch 区域
36 * @param remote_hartindex: 目标 Hart 的 hart index
37 * @param event: IPI 事件类型（bit 编号）
38 * @param data: 传递给 IPI 事件处理函数的数据
39 */
40static int sbi_ipi_send(struct sbi_scratch *scratch, u32 remote_hartindex,
41			u32 event, void *data)
42{
43	// ...
44
45	/*
46	 * 在远程 Hart 的 scratch 区域设置 IPI 类型并触发中断
47	 *
48	 * 多个 Hart 可能同时向同一个远程 Hart 发送 IPI
49	 * 使用原子操作确保只有一个 Hart 触发底层硬件发送
50	 * linux 调用下来使用的是 ipi_smode_event
51	 * 只有当 ipi_data->ipi_type 为 0, 没有事件需要处理时，才能发
52	 */
53	if (!__atomic_fetch_or(&ipi_data->ipi_type, BIT(event), __ATOMIC_RELAXED))
54		ret = sbi_ipi_raw_send(remote_hartindex);
55
56	// ...
57}
58
59/*
60 * 底层 IPI 发送函数
61 * 调用硬件设备驱动的 ipi_send 回调
62 *
63 * @param hartindex: 目标 Hart 的 hart index
64 */
65int sbi_ipi_raw_send(u32 hartindex)
66{
67	if (!ipi_dev || !ipi_dev->ipi_send)
68		return SBI_EINVAL;
69
70	/*
71	 * 内存屏障：确保在此函数之前完成的内存或 MMIO 写入
72	 * 不会在 ipi_send() 设备回调的内存或 MMIO 写入之后被观察到
73	 */
74	wmb();
75
76	ipi_dev->ipi_send(hartindex);
77	return 0;
78}

4.4 IPI 事件处理

 1// lib/sbi/sbi_ipi.c
 2
 3/**
 4 * 处理当前 Hart 的所有待处理 IPI 事件
 5 * 在 M 模式处理完 IPI 后返回 S 模式之前调用
 6 */
 7void sbi_ipi_process(void)
 8{
 9	unsigned long ipi_type;
10	unsigned int ipi_event;
11	const struct sbi_ipi_event_ops *ipi_ops;
12	struct sbi_scratch *scratch = sbi_scratch_thishart_ptr();
13	struct sbi_ipi_data *ipi_data =
14			sbi_scratch_offset_ptr(scratch, ipi_data_off);
15
16	sbi_pmu_ctr_incr_fw(SBI_PMU_FW_IPI_RECVD);
17
18	/* 清除硬件 IPI 状态 */
19	sbi_ipi_raw_clear();
20
21	/* 原子地交换出所有待处理的 IPI 类型并清零 */
22	ipi_type = atomic_raw_xchg_ulong(&ipi_data->ipi_type, 0);
23
24	/* 遍历所有 IPI 事件类型，处理每个待处理的 IPI */
25	ipi_event = 0;
26	while (ipi_type) {
27		if (ipi_type & 1UL) {
28			/* 调用对应事件类型的处理函数 */
29			ipi_ops = ipi_ops_array[ipi_event];
30			if (ipi_ops)
31				ipi_ops->process(scratch);
32		}
33		ipi_type = ipi_type >> 1;
34		ipi_event++;
35	}
36}
37
38/**
39 * S 模式 IPI 事件处理函数
40 * 设置 S 模式软件中断挂起位（SSIP），通知目标 Hart 有 IPI 待处理
41 */
42static void sbi_ipi_process_smode(struct sbi_scratch *scratch)
43{
44	/* 设置 mip.SSIP（S 模式软件中断挂起位） */
45	csr_set(CSR_MIP, MIP_SSIP);
46}
47
48/**
49 * Halt IPI 事件处理函数
50 * 请求目标 Hart 进入休眠状态
51 */
52static void sbi_ipi_process_halt(struct sbi_scratch *scratch)
53{
54	sbi_hsm_hart_stop(scratch, true);
55}

4.5 IPI 初始化

 1// lib/sbi/sbi_ipi.c
 2
 3/*
 4 * IPI 子系统初始化
 5 *
 6 * @param scratch: 当前 Hart 的 scratch 区域
 7 * @param cold_boot: 是否是冷启动（true 表示首次启动，false 表示从 S 模式返回）
 8 *
 9 * @return 0 表示成功，负数表示错误码
10 */
11int sbi_ipi_init(struct sbi_scratch *scratch, bool cold_boot)
12{
13	int ret;
14	struct sbi_ipi_data *ipi_data;
15
16	if (cold_boot) {
17		/* 分配 scratch 空间用于存储 IPI 数据 */
18		ipi_data_off = sbi_scratch_alloc_offset(sizeof(*ipi_data));
19		if (!ipi_data_off)
20			return SBI_ENOMEM;
21
22		/* 注册 S 模式 IPI 事件 */
23		ret = sbi_ipi_event_create(&ipi_smode_ops);
24		if (ret < 0)
25			return ret;
26		ipi_smode_event = ret;
27
28		/* 注册 Halt IPI 事件 */
29		ret = sbi_ipi_event_create(&ipi_halt_ops);
30		if (ret < 0)
31			return ret;
32		ipi_halt_event = ret;
33
34		/* 初始化平台 IPI 支持（解析设备树并注册 IPI 设备） */
35		ret = sbi_platform_ipi_init(sbi_platform_ptr(scratch));
36		if (ret)
37			return ret;
38	}
39
40	/* 初始化当前 Hart 的 IPI 数据 */
41	ipi_data = sbi_scratch_offset_ptr(scratch, ipi_data_off);
42	ipi_data->ipi_type = ATOMIC_INIT(0);
43
44	/* 清除当前 Hart 的待处理 IPI */
45	sbi_ipi_raw_clear();
46
47	/* 使能 M 模式软件中断 (M 模式下的 MSIP) */
48	csr_set(CSR_MIE, MIP_MSIP);
49
50	return 0;
51}

5. Andes PLICSW 设备驱动

PLICSW（PLIC Software） 是 Andes Technology（晶心科技） 提供的软件中断解决方案，作为 PLIC（Platform Level Interrupt Controller，平台级中断控制器） 的扩展。

5.1 设备驱动注册

 8	{ },
 9};

5.2 驱动实现

 1// lib/utils/ipi/andes_plicsw.c
 2
 3/*
 4 * PLICSW IPI 发送函数
 5 * 通过设置 pending 寄存器触发目标 Hart 的软件中断
 6 *
 7 * @param hart_index: 目标 Hart 的 hart index
 8 */
 9static void plicsw_ipi_send(u32 hart_index)
10{
11	ulong pending_reg;
12	u32 interrupt_id, word_index, pending_bit;
13	u32 target_hart = sbi_hartindex_to_hartid(hart_index);
14
15	if (plicsw.hart_count <= target_hart)
16		ebreak();
17
18	/*
19	 * 每个 Hart 分配一个 bit
20	 * Bit 0 硬连线到 0，不可使用
21	 * Bit (X+1) 表示 IPI 发送到 Hart X
22	 */
23	interrupt_id = target_hart + 1;
24	word_index   = interrupt_id / 32;
25	pending_bit  = interrupt_id % 32;
26	pending_reg  = plicsw.addr + PLICSW_PENDING_BASE + word_index * 4;
27
28	/*
29	 * 写 1 到对应 bit 触发目标 Hart 的 mip.MSIP
30	 * MSIP (Machine Software Interrupt Pending) 是 M 模式软件中断挂起位
31	 */
32	writel_relaxed(BIT(pending_bit), (void *)pending_reg);
33}
34
35/* 注册 PLICSW 设备 */
36static struct sbi_ipi_device plicsw_ipi = {
37	.name      = "andes_plicsw",
38	.ipi_send  = plicsw_ipi_send,
39	.ipi_clear = plicsw_ipi_clear
40};

6. SBI IPI 扩展接口

6.1 ECALL 处理

 1// lib/sbi/sbi_ecall_ipi.c
 2
 3/*
 4 * SBI IPI 扩展 ECALL 处理函数
 5 * 处理来自 S 模式的 IPI 相关调用
 6 *
 7 * @param extid: 扩展 ID（SBI_EXT_IPI）
 8 * @param funcid: 函数 ID
 9 * @param regs: 触发 ECALL 时的寄存器状态
10 * @param out: ECALL 返回值结构体
11 *
12 * @return 0 表示成功，负数表示错误码
13 */
14static int sbi_ecall_ipi_handler(unsigned long extid, unsigned long funcid,
15				 struct sbi_trap_regs *regs,
16				 struct sbi_ecall_return *out)
17{
18	int ret = 0;
19
20	if (funcid == SBI_EXT_IPI_SEND_IPI) {
21		/*
22		 * 发送 IPI 到 S 模式
23		 * a0: hmask - 目标 Hart 位掩码
24		 * a1: hbase - 目标 Hart 的起始 Hart ID
25		 */
26		ret = sbi_ipi_send_smode(regs->a0, regs->a1);
27	} else {
28		ret = SBI_ENOTSUPP;
29	}
30
31	return ret;
32}
33
34/* 注册 SBI IPI 扩展 */
35struct sbi_ecall_extension ecall_ipi = {
36	.name			= "ipi",
37	.extid_start		= SBI_EXT_IPI,       // 0x73504900 ('SPI' << 8)
38	.extid_end		= SBI_EXT_IPI,
39	.register_extensions	= sbi_ecall_ipi_register_extensions,
40	.handle			= sbi_ecall_ipi_handler,
41};
