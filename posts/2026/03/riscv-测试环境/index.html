riscv 测试环境

1. 构建 riscv 架构运行的 qemu

基于 riscv 的 ubuntu rootfs，编译 qemu，riscv 的 rootfs构建方式如下：

‣

 1# 1. 进入 riscv rootfs
 2sudo chroot your_riscv_ubuntu_rootfs /bin/bash
 3
 4# 2. 安装编译 qemu 的必要库
 5apt install python3-pip ninja-build build-essential autoconf automake libtool pkg-config libblkid-dev gettext bison flex 
 6
 7# 3. 安装静态编译 qemu 的 libmount.a
 8apt source util-linux
 9cd ./build-mount/util-linux-2.39.3
10./configure --prefix=/usr/local --enable-static --disable-shared --disable-all-programs --enable-libmount --enable-libblkid --with-pic CFLAGS="-fvisibility=hidden"
11make -j8
12make install
13
14# 4. 编译 qemu
15cd qemu_dir
16sed -i "s/imply\ TEST_DEVICES/imply\ TEST_DEVICES\n\ \ \ \ imply\ IOMMUFD/g" ./hw/riscv/Kconfig # virt 机器添加 iommufd 的支持
17grep -rl "crc32c(" . | xargs sed -i 's/crc32c(/qemu_crc32c(/g'mkdir build # 防止 qemu 中定义的 crc32c 函数和 libblkid 中定义的 crc32 函数重复定义 
18cd build
19KERNEL_HDR_PATH=/home/jinglin/lrzx_workspace/linux-next/initfs/linux_header ../configure --target-list=riscv64-softmmu --enable-kvm --enable-iommufd --static --prefix=/opt/
20make -j8
21make install

2. 测试

2.1 修改内核

0001-riscv-iommu-support-vfio-to-riscv.patch

2.2 配置内核

 1# 1. 配置 kvm
 2CONFIG_KVM=m
 3
 4# 2. 配置 iommu
 5CONFIG_IOMMU_SUPPORT=y
 6CONFIG_RISCV_IOMMU=y
 7CONFIG_RISCV_IOMMU_PCI=y
 8
 9# 3. 配置 vfio
10CONFIG_VFIO=y
11CONFIG_VFIO_PCI=y # vfio-pci 设备驱动
12CONFIG_KVM_VFIO=y # 目前 riscv 没有支持，需要修改代码
13
14# 3. 配置 iommufd，通过用户态接口控制 vfio
15CONFIG_IOMMUFD=y
16CONFIG_VFIO_DEVICE_CDEV=y # 用于 iommufd
17CONFIG_VFIO_NOIOMMU=y
18
19# 4. 配置旧版 vfio 接口
20CONFIG_VFIO_CONTAINER=y   # 用于 IOMMU type1，目前 riscv 没有支持，需要修改代码
21CONFIG_VFIO_IOMMU_TYPE1=y

2.3 安装驱动

1# 挂载文件系统
2sudo mount -t ext2 ./your_rootfs.img /tmp
3
4# 安装头文件 和 驱动
5sudo make INSTALL_MOD_PATH=/tmp modules_install
6sudo make INSTALL_HDR_PATH=/tmp/usr headers_install

2.4 测试 kvm

2.4.1 启动 host

 1# qemu 启动带 H 扩展的 cpu，virt 的cpu 默认带 H 扩展，或者显性使用 h=true
 2qemu-system-riscv64 -nographic -M virt,aia=aplic-imsic,aia-guests=4 -cpu rv64,h=true -m 2G \
 3        -device riscv-iommu-pci,addr=01.0,intremap=on \
 4        -device pcie-root-port,id=rp1,bus=pcie.0,chassis=1,slot=1,addr=02.0 \
 5        -device e1000e,netdev=net0,bus=rp1,addr=00.0,mac=52:54:00:12:34:56 \
 6        -netdev user,id=net0 \
 7        -kernel arch/riscv/boot/Image -append "root=/dev/vda rw console=ttyS0" \
 8        -drive if=none,file=./initramfs.cpio.test,format=raw,id=hd0 \
 9        -device virtio-blk-device,drive=hd0 \
10        -monitor telnet:localhost:5678,server,nowait

2.4.2 启动 guest

1# 1. 安装 kvm.ko
2insmod /lib/modules/6.19.0-rc1-dirty/kernel/arch/riscv/kvm/kvm.ko
3
4# 2. 启动虚拟机
5/opt/bin/qemu-system-riscv64 -nographic -M virt -m 1G -accel kvm \
6        -monitor telnet:localhost:5678,server,nowait \
7        -kernel ./Image -initrd ./initramfs_24M.cpio \
8        -append "root=/dev/ram0 console=ttyS0"

2.5 测试 iommu

说明文档：linux/Documentation/driver-api/vfio.rst

    `qemu/docs/specs/riscv-iommu.rst,支持 iommu-sys 和 iommu-pci 两种模式` 

    `qemu/docs/devel/vfio-iommufd.rst`

2.5.1 启动 host

 1# qemu 启动带 H 扩展的 cpu，virt 的cpu 默认带 H 扩展，或者显性使用 h=true
 2# 可以使用命令行 iommu.passthrough=0 riscv_iommu.force_isolation=1 强制开启 mmu
 3# iommu.passthrough=0：禁用物理透传，强制启用 DMA 地址转换。
 4# riscv_iommu.force_isolation=1：强制为每个设备创建独立的 IOMMU 域, 这是 amd 的参数
 5# iommufd.allow_unsafe_interrupts=1 vfio_iommu_type1.allow_unsafe_interrupts=1 riscv vfio 支持不完善，需要强制允许 不安全中断
 6qemu-system-riscv64 -nographic -M virt,aia=aplic-imsic,aia-guests=4,iommu-sys=on
 7				-cpu rv64,h=true -m 2G \
 8        -device e1000e,netdev=net0,mac=52:54:00:12:34:56 \
 9        -netdev user,id=net0,net=10.0.2.0/24,dhcpstart=10.0.2.11 \
10        -kernel arch/riscv/boot/Image -append "root=/dev/vda rw console=ttyS0 iommufd.allow_unsafe_interrupts=1 vfio_iommu_type1.allow_unsafe_interrupts=1" \
11        -drive if=none,file=./initramfs.cpio.test,format=raw,id=hd0 \
12        -device virtio-blk-device,drive=hd0 \
13        -monitor telnet:localhost:5678,server,nowait
14        
15# 查看有没有成功注册 iommu
16dmesg | grep -i iommu
17[    0.335548][    T1] iommu: Default domain type: Translated
18[    0.335614][    T1] iommu: DMA domain TLB invalidation policy: strict mode

20[    2.288427][   T42] pci 0000:00:00.0: Adding to iommu group 0
21[    2.289751][   T42] pci 0000:00:01.0: Adding to iommu group 1
22
23# 检测设备由 iommu 隔离，目录存在表示，设备挂在 iommu 的隔离组上
24ls /sys/bus/pci/devices/0000\:00\:01.0/iommu_group/
